clc
clear
close all

cajita3d

xlim([-x x])
ylim([-y y])
zlim([-z z])

axis equal
grid on
hold on
view(45,30)

thetaX = input("Cuanto quieres rotar en X [rad]: ");
thetaY = input("Cuanto quieres rotar en Y [rad]: ");
thetaZ = input("Cuanto quieres rotar en Z [rad]: ");

for t = 0:0.02:1

    Rx = Rotaciones('x',t*thetaX);
    Ry = Rotaciones('y',t*thetaY);
    Rz = Rotaciones('z',t*thetaZ);

    R = Rx*Ry*Rz;

    joint1R = R*joint1;
    joint2R = R*joint2;
    joint3R = R*joint3;
    joint4R = R*joint4;
    joint5R = R*joint5;
    joint6R = R*joint6;
    joint7R = R*joint7;
    joint8R = R*joint8;

    cla

    grid on
    hold on

    line([-x x],[0 0],[0 0], ...
        'LineWidth',1.5,'Color','red');

    line([0 0],[-y y],[0 0], ...
        'LineWidth',1.5,'Color','magenta');

    line([0 0],[0 0],[-z z], ...
        'LineWidth',1.5,'Color','green');

    xlim([-x x])
    ylim([-y y])
    zlim([-z z])

    axis equal
    view(45,30)

    line([joint1R(1) joint2R(1)], ...
         [joint1R(2) joint2R(2)], ...
         [joint1R(3) joint2R(3)], ...
         'LineWidth',3,'Color','cyan');

    line([joint1R(1) joint3R(1)], ...
         [joint1R(2) joint3R(2)], ...
         [joint1R(3) joint3R(3)], ...
         'LineWidth',3,'Color','cyan');

    line([joint2R(1) joint4R(1)], ...
         [joint2R(2) joint4R(2)], ...
         [joint2R(3) joint4R(3)], ...
         'LineWidth',3,'Color','cyan');

    line([joint3R(1) joint4R(1)], ...
         [joint3R(2) joint4R(2)], ...
         [joint3R(3) joint4R(3)], ...
         'LineWidth',3,'Color','cyan');

    line([joint1R(1) joint5R(1)], ...
         [joint1R(2) joint5R(2)], ...
         [joint1R(3) joint5R(3)], ...
         'LineWidth',3,'Color','cyan');

    line([joint2R(1) joint6R(1)], ...
         [joint2R(2) joint6R(2)], ...
         [joint2R(3) joint6R(3)], ...
         'LineWidth',3,'Color','cyan');

    line([joint3R(1) joint7R(1)], ...
         [joint3R(2) joint7R(2)], ...
         [joint3R(3) joint7R(3)], ...
         'LineWidth',3,'Color','cyan');

    line([joint4R(1) joint8R(1)], ...
         [joint4R(2) joint8R(2)], ...
         [joint4R(3) joint8R(3)], ...
         'LineWidth',3,'Color','cyan');

    line([joint5R(1) joint6R(1)], ...
         [joint5R(2) joint6R(2)], ...
         [joint5R(3) joint6R(3)], ...
         'LineWidth',3,'Color','cyan');

    line([joint6R(1) joint8R(1)], ...
         [joint6R(2) joint8R(2)], ...
         [joint6R(3) joint8R(3)], ...
         'LineWidth',3,'Color','cyan');

    line([joint8R(1) joint7R(1)], ...
         [joint8R(2) joint7R(2)], ...
         [joint8R(3) joint7R(3)], ...
         'LineWidth',3,'Color','cyan');

    line([joint7R(1) joint5R(1)], ...
         [joint7R(2) joint5R(2)], ...
         [joint7R(3) joint5R(3)], ...
         'LineWidth',3,'Color','cyan');

    drawnow
    pause(0.02)

end
